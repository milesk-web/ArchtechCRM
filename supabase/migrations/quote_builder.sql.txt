-- Quote builder: catalogue tables + quote_lines
-- No commercial data is seeded here -- every catalogue table is created
-- empty, to be populated later from the real pricing source. unit_cost/rate
-- columns are nullable specifically so a catalogue entry can exist (e.g. a
-- profile name) before its price is known -- the UI is expected to show "—"
-- for anything with a NULL price, not a fabricated $0.

-- ASSUMPTION FLAGGED: this migration enables RLS with a permissive
-- "authenticated users can do everything" policy on each new table, since I
-- can't see the existing RLS conventions used on your `jobs`/`quotes`
-- tables from here. If your project uses a different pattern (e.g.
-- per-organisation row scoping), these policies should be adjusted to match
-- before relying on this in production -- please confirm/adjust rather than
-- assume this is correct as-is.

create table if not exists profiles (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  section text not null,               -- e.g. 'Roofing' or 'Wall Cladding'
  measurement_type text not null check (measurement_type in ('gauge', 'width')),
  active boolean not null default true,
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists profile_options (
  id uuid primary key default gen_random_uuid(),
  profile_id uuid not null references profiles(id) on delete cascade,
  value text not null,
  active boolean not null default true,
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists materials (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  active boolean not null default true,
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists material_colours (
  id uuid primary key default gen_random_uuid(),
  material_id uuid not null references materials(id) on delete cascade,
  name text not null,
  active boolean not null default true,
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists underlays (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  unit text not null,
  unit_cost numeric,                   -- nullable -- populate later
  active boolean not null default true,
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists flashing_types (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  unit text not null,
  unit_cost numeric,
  active boolean not null default true,
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists labour_types (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  unit text not null,
  rate numeric,
  active boolean not null default true,
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists accessories (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  unit text not null,
  unit_cost numeric,
  active boolean not null default true,
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists quote_lines (
  id uuid primary key default gen_random_uuid(),
  quote_id uuid not null references quotes(id) on delete cascade,
  section text not null,               -- 'Roofing' | 'Wall Cladding' | 'Accessories'
  category text not null,              -- 'Material' | 'Flashing' | 'Labour' | 'Accessory'
  description text not null default '',
  quantity numeric not null default 0 check (quantity >= 0),
  unit text not null default 'item',
  unit_cost numeric,                   -- nullable -- "—" in the UI until known
  sell_price numeric,                  -- nullable -- reserved for markup phase
  sort_order integer not null default 0,
  metadata jsonb not null default '{}'::jsonb,  -- selected profile/material/colour/etc.
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index if not exists quote_lines_quote_id_idx on quote_lines(quote_id);
create index if not exists profile_options_profile_id_idx on profile_options(profile_id);
create index if not exists material_colours_material_id_idx on material_colours(material_id);
